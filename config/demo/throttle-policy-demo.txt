# ---------------------------------------------------------------
# DEMO Throttle Policy (Aggressively small bandwidth limits)
# These small limits ensure the small demo PCAP bursts past
# the Token Bucket capacity immediately.
#
# 1 Kbps limit = 125 bytes/sec burst capacity
# 2 Kbps limit = 250 bytes/sec burst capacity
# ---------------------------------------------------------------

LIMIT_ESSENTIAL=-1
LIMIT_STANDARD=2
LIMIT_ENTERTAINMENT=1

# At 1-2 Kbps a 600-byte demo packet needs 1.4-3.8 s to get through.
# A longer queue limit lets the demo show those packets as DELAY instead of DROP.
# (Real setting: config/throttle-policy.txt uses the 500 ms default.)
MAX_QUEUE_DELAY_MS=5000

# App Mappings
WHATSAPP=ESSENTIAL
GPAY=ESSENTIAL
GOOGLE=STANDARD
GMAIL=STANDARD
NETFLIX=ENTERTAINMENT
SPOTIFY=ENTERTAINMENT
